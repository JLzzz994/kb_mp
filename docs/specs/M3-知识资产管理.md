class TxtParser(BaseParser):
    def parse(self, path: Path) -> str:
        try:
            return path.read_text(encoding="utf-8")
        except UnicodeDecodeError:
            return path.read_text(encoding="gbk")  # Windows 中文 TXT 兼容

### 5.6 KnowledgePermissionService（**核心**）

```python
class KnowledgePermissionService:
    async def configure(self, unit_id: int, req: ConfigurePermissionsRequest, user: CurrentUser) -> list[PermissionEntryResponse]:
        # 1) 校验至少一条权限
        # 2) DELETE unit_permissions WHERE unit_id
        # 3) INSERT 批量
        # 4) 不需要失效 Redis 位图（数据权限不影响 ai:chat，仅影响 unit 可读性）
        # 5) 返回聚合权限视图（含 target_label JOIN）

    async def load_user_permission_bitmap(self, user_id: int) -> UserPermissionBitmap:
        """M1 AuthService 登录时调用：返回用户可访问的 unit_ids 集合（全集）"""
        # 1) SELECT target_id FROM unit_permissions WHERE target_type='global'
        # 2) UNION department ids：SELECT unit_id FROM unit_permissions up
        #    JOIN user_department_paths udp ON up.target_id = udp.dept_id
        # 3) UNION role ids：SELECT unit_id FROM unit_permissions up
        #    JOIN user_roles ur ON up.target_id = ur.role_id WHERE ur.user_id = ?
        # 4) UNION user ids：SELECT unit_id WHERE target_id = user_id AND target_type='user'
        # 缓存 Redis：auth:bitmap:{user_id} → JSON(unit_ids)

    async def check_permissions(self, user_id: int, unit_ids: list[int]) -> CheckPermissionsResponse:
        """M3 定义，M4 调用；按 Redis 位图优先"""
        bitmap = await self._redis.get_bitmap(user_id) or await self._load_and_cache(user_id)
        authorized = [uid for uid in unit_ids if uid in bitmap]
        unauthorized = [uid for uid in unit_ids if uid not in bitmap]
        return CheckPermissionsResponse(authorized_unit_ids=authorized, unauthorized_unit_ids=unauthorized)

    def compute_user_permission_bitmap_sync(
        self, user: CurrentUser, unit_permissions: list[UnitPermissionRecord]
    ) -> set[int]:
        """纯函数：内存集合运算（无 IO），供 LangGraph permission_filter 节点调用"""
        global_units = {up.unit_id for up in unit_permissions if up.target_type == "global"}
        dept_units = {up.unit_id for up in unit_permissions if up.target_type == "department" and up.target_id in user.dept_ids}
        role_units = {up.unit_id for up in unit_permissions if up.target_type == "role" and up.target_id in user.role_ids}
        user_units = {up.unit_id for up in unit_permissions if up.target_type == "user" and up.target_id == user.id}
        return global_units | dept_units | role_units | user_units
```

## 8. 异步与适配器

| 项 | 描述 |
| --- | --- |
| 解析器 | pypdf / python-docx / markdown-it；UTF-8 TXT 失败回退 GBK |
| 切片器 | `langchain_text_splitters.RecursiveCharacterTextSplitter`，参数见 §5.4 |
| Embedding | OpenAI `text-embedding-3-small`，dim=1536 |
| 向量库 | Milvus 单集合 `kb_units`：`id / unit_id / title / content / embedding / category`；索引 `embedding HNSW COSINE M=16 efConstruction=64` |
| 写入路径 | SQL INSERT 后 `BackgroundTasks.add_task(milvus_upsert, chunks)`；失败时 `UPDATE status='vector_pending'` + 日志 |
| 删除路径 | SQL DELETE + `milvus_delete(ids)` 同事务边界外（失败仅日志） |
| 文件存储 | 本地 `storage/uploads/{uuid}.{ext}`，DB 存 `source_file_name` + `file_size` |
