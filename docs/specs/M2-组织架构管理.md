```python
# department_schema.py
class DepartmentNode(BaseModel):
    id: int
    name: str
    parent_id: int | None
    leader_id: int | None
    member_count: int
    children: list[DepartmentNode] = []


class DepartmentCreate(BaseModel):
    name: str = Field(min_length=1, max_length=64)
    parent_id: int | None = None
    leader_id: int | None = None
    sort_order: int = 0


# user_schema.py
class UserCreate(BaseModel):
    username: str = Field(min_length=3, max_length=64, pattern=r"^[a-zA-Z0-9_]+$")
    password: str = Field(min_length=8, max_length=128)
    display_name: str = Field(min_length=1, max_length=64)
    department_id: int
    role_ids: list[int] = Field(min_length=1)


class UserResponse(BaseModel):
    model_config = ConfigDict(from_attributes=True)
    id: int
    username: str
    display_name: str
    department_id: int
    department_name: str
    role_codes: list[str]
    status: int
    created_at: datetime
    updated_at: datetime


class UserListResponse(Page[UserResponse]): ...


class ResetPasswordRequest(BaseModel):
    new_password: str = Field(min_length=8, max_length=128)


# role_schema.py
class RoleResponse(BaseModel):
    id: int
    role_name: str
    role_code: str
    description: str | None
    permissions: list[str]


class AssignPermissionsRequest(BaseModel):
    permission_codes: list[str] = Field(min_length=1)
    permission_type: Literal["menu", "button", "api"] = "menu"
```

## 5. 后端设计

### 5.1 路由（`org_router.py`）

**路由前缀**：所有端点统一 `/api/v1/<domain>` 前缀，详见接口约定文档 §7 各模块接口分组清单。

| Method | 完整路径 | 权限 | Service 方法 |
| --- | --- | --- | --- |
| GET | `/api/v1/org/departments` | `dept:read` | `DepartmentService.list_tree()` |
| POST | `/api/v1/org/departments` | `dept:write` | `DepartmentService.create()` |
| PUT | `/api/v1/org/departments/{id}` | `dept:write` | `DepartmentService.update()` |
| DELETE | `/api/v1/org/departments/{id}` | `dept:write` | `DepartmentService.delete()` |
| GET | `/api/v1/org/users` | `user:read` | `UserService.list()` |
| GET | `/api/v1/org/users/{id}` | `user:read` | `UserService.get()` |
| POST | `/api/v1/org/users` | `user:write` | `UserService.create()` |
| PUT | `/api/v1/org/users/{id}` | `user:write` | `UserService.update()` |
| PATCH | `/api/v1/org/users/{id}/status` | `user:write` | `UserService.set_status()` |
| POST | `/api/v1/org/users/{id}/reset-password` | `user:write` | `UserService.reset_password()` |
| GET | `/api/v1/org/roles` | `role:read` | `RoleService.list()` |
| POST | `/api/v1/org/roles/{id}/permissions` | `role:write` | `RoleService.assign_permissions()` |
| GET | `/api/v1/org/permissions` | `role:read` | `RoleService.list_all_permission_codes()` |

### 5.2 Service 签名（关键方法）

```python
class DepartmentService:
    async def list_tree(self) -> list[DepartmentNode]: ...  # 一次 SQL 拉全表 + 内存组装树
    async def create(self, data: DepartmentCreate) -> DepartmentNode: ...  # 循环引用校验 parent_id
    async def delete(self, dept_id: int) -> None: ...  # 有子部门或成员时抛 DeptNotEmptyError


class UserService:
    async def list(
        self,
        *,
        page: int,
        page_size: int,
        department_id: int | None,
        status: int | None,
        keyword: str | None,
    ) -> tuple[list[UserResponse], int]: ...
    async def create(self, data: UserCreate) -> UserResponse: ...  # 哈希密码 + 插入 + 绑角色
    async def reset_password(
        self, user_id: int, req: ResetPasswordRequest
    ) -> None: ...  # 哈希 + UPDATE
    async def set_status(self, user_id: int, status: int) -> None: ...  # 0/1；停用后踢出位图


class RoleService:
    async def list(self) -> list[RoleResponse]: ...  # 包含 permissions
    async def assign_permissions(self, role_id: int, req: AssignPermissionsRequest) -> None:
        ...
        # 1) 删旧 2) 批量 INSERT 3) 找出所有持此 role 的 user_id 4) DEL auth:bitmap:{user_id}

    async def list_all_permission_codes(self) -> list[str]: ...  # 返回 §4.2 M1 §10.1 中的 17 个码
```

### 5.5 事务与幂等
- 用户创建：username UQ 冲突由 DB 抛 `IntegrityError` → 捕获转 `UsernameConflictError`
- 角色权限分配：先 `DELETE` 再批量 `INSERT`，单事务内完成
- 部门删除：前置校验子部门 / 成员；非空抛错

### 6.3 关键状态

| 状态 | UI 反馈 |
| --- | --- |
| 用户名重复 | UserFormDialog 内联错误：username 字段红框 + "用户名已存在" |
| 删除非空部门 | DepartmentFormDialog 操作按钮 disabled + tooltip "该部门含子部门或成员" |
| 停用用户 | StatusConfirmDialog 二次确认（含影响说明：关联会话将失效） |
| 权限保存中 | 全屏 loading；保存成功后跳列表并 toast |
| 权限树懒加载 | 仅渲染当前展开层级，折叠收起 |

### 6.4 API 方法

```typescript
// frontend/src/api/org.ts
export const orgApi = {
  listDepartments: () => api.get<DepartmentNode[]>("/org/departments").then(r => r.data),
  listUsers: (params: UserListParams) => api.get<UserListResponse>("/org/users", { params }).then(r => r.data),
  createUser: (data: UserCreate) => api.post<UserResponse>("/org/users", data).then(r => r.data),
  resetPassword: (userId: number, data: ResetPasswordRequest) => api.post(`/org/users/${userId}/reset-password`, data),
  listRoles: () => api.get<RoleResponse[]>("/org/roles").then(r => r.data),
  assignPermissions: (roleId: number, data: AssignPermissionsRequest) => api.post(`/org/roles/${roleId}/permissions`, data),
  listAllPermissionCodes: () => api.get<string[]>("/org/permissions").then(r => r.data),
};
```

### 9.2 关键事件日志
| 事件 | 字段 |
| --- | --- |
| 用户创建 | log:info `event=user.create actor_id=... target_id=...` |
| 密码重置 | log:warn `event=user.reset_password actor_id=... target_id=...` |
| 角色权限变更 | log:warn `event=role.permissions_change actor_id=... role_id=... affected_users=N` |
| 部门删除 | log:warn `event=department.delete actor_id=... dept_id=...` |

### 9.3 脱敏
- 密码字段永不入日志
- 操作日志保留 actor_id 与 target_id，不含明文内容
